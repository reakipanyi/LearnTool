private string GetFillPromptScript(string? host, string prompt)
{
    if (string.IsNullOrEmpty(host))
        return "/* unknown host */";

    var escapedPrompt = EscapeForJavaScript(prompt);

    // 更通用的脚本，尝试多种选择器
    return $@"(function(){{
        var selectors = [
            'textarea',
            'textarea[placeholder]',
            'textarea[placeholder*="输入"]',
            'textarea[placeholder*="问题"]',
            'input[type="text"]',
            '[contenteditable="true"]',
            '[contenteditable]',
            '.chat-input textarea',
            '#chat-input textarea',
            '.prompt-textarea',
            '[class*="input"] textarea'
        ];
        
        for (var i = 0; i < selectors.length; i++) {{
            var t = document.querySelector(selectors[i]);
            if (t) {{
                t.value = '{escapedPrompt}';
                t.innerText = '{escapedPrompt}';
                t.textContent = '{escapedPrompt}';
                t.dispatchEvent(new Event('input', {{bubbles:true}}));
                t.dispatchEvent(new Event('change', {{bubbles:true}}));
                t.dispatchEvent(new KeyboardEvent('keydown', {{bubbles:true}}));
                t.focus();
                return 'filled: ' + selectors[i];
            }}
        }}
        
        // 如果找不到，尝试点击输入区域后再填充
        var inputAreas = document.querySelectorAll('[class*="input"], [class*="chat"]');
        for (var j = 0; j < inputAreas.length; j++) {{
            inputAreas[j].click();
        }}
        
        return 'not found';
    }})();";
}

// 3. 在 OnNavigationCompleted 中增加延迟和重试机制
private async void OnNavigationCompleted(object? sender, CoreWebView2NavigationCompletedEventArgs e)
{
    if (_webView == null || !e.IsSuccess) return;

    SetStatus("页面加载完成");
    var host = _webView.Source?.Host;

    if (!string.IsNullOrEmpty(_pendingPrompt))
    {
        // 增加延迟，等待页面完全渲染
        await Task.Delay(1000);
        
        int retryCount = 0;
        const int maxRetry = 3;
        bool filled = false;
        
        while (!filled && retryCount < maxRetry)
        {
            try
            {
                SetStatus($"正在填充提示词... (尝试 {retryCount + 1}/{maxRetry})");
                string script = GetFillPromptScript(host, _pendingPrompt);
                var result = await _webView.CoreWebView2.ExecuteScriptAsync(script);
                
                if (result.Contains("filled"))
                {
                    filled = true;
                    SetStatus("就绪");
                }
                else
                {
                    retryCount++;
                    await Task.Delay(500);
                }
            }
            catch (Exception ex)
            {
                retryCount++;
                _logger?.LogError(ex, "填充提示词失败，尝试 {RetryCount}", retryCount);
                await Task.Delay(500);
            }
        }
        
        if (!filled)
        {
            ShowError("无法自动填充提示词，请手动输入");
        }
        
        _pendingPrompt = null;
    }
}

// 创建 AIPromptConfig.cs 模型类
public class AIPromptConfig
{
    public PromptCategories Prompts { get; set; } = new();
    public Dictionary<string, string> SystemPrompts { get; set; } = new();
}

public class PromptCategories
{
    public LanguagePrompts Explanation { get; set; } = new();
    public Dictionary<string, string> QuickActions { get; set; } = new();
    public LanguagePrompts Exercise { get; set; } = new();
    public string Summarize { get; set; } = "";
}

public class LanguagePrompts
{
    public Dictionary<string, string> Chinese { get; set; } = new();
    public Dictionary<string, string> English { get; set; } = new();
}

// 创建 AIPromptService.cs 服务类
public class AIPromptService
{
    private readonly AIPromptConfig _config;
    
    public AIPromptService(string configPath)
    {
        var json = File.ReadAllText(configPath);
        _config = JsonConvert.DeserializeObject<AIPromptConfig>(json) 
                  ?? new AIPromptConfig();
    }
    
    public string GetExplanationPrompt(string text, string language, string subType)
    {
        var prompts = language == "中文" ? _config.Prompts.Explanation.Chinese 
                                            : _config.Prompts.Explanation.English;
        
        if (prompts.TryGetValue(subType, out var template))
        {
            return template.Replace("{text}", text);
        }
        
        // 默认提示词
        return $"请简要解释：{text}";
    }
    
    public string GetQuickActionPrompt(string action, string context)
    {
        if (_config.Prompts.QuickActions.TryGetValue(action, out var template))
        {
            return $"{template}\n{context}";
        }
        return context;
    }
    
    public string GetExercisePrompt(string text, string language)
    {
        var prompts = language == "中文" ? _config.Prompts.Exercise.Chinese 
                                           : _config.Prompts.Exercise.English;
        return prompts.Replace("{text}", text);
    }
}

// 创建 SubjectTemplateService.cs
public class SubjectTemplateService
{
    private readonly Dictionary<string, SubjectConfig> _subjects;
    
    public SubjectTemplateService(string configPath)
    {
        var json = File.ReadAllText(configPath);
        var config = JsonConvert.DeserializeObject<SubjectTemplateConfig>(json);
        _subjects = config?.Subjects ?? new Dictionary<string, SubjectConfig>();
    }
    
    public List<string> GetAvailableSubjects()
    {
        return _subjects.Keys.ToList();
    }
    
    public List<string> GetCategoriesForSubject(string subject)
    {
        if (_subjects.TryGetValue(subject, out var subjectConfig))
        {
            return subjectConfig.Categories;
        }
        return new List<string>();
    }
    
    public List<string> GetFieldsForCategory(string subject, string category)
    {
        if (_subjects.TryGetValue(subject, out var subjectConfig))
        {
            if (subjectConfig.Templates.TryGetValue(category, out var template))
            {
                return template.Fields;
            }
        }
        return new List<string>();
    }
    
    public string GetJsonTemplate(string subject, string category)
    {
        if (_subjects.TryGetValue(subject, out var subjectConfig))
        {
            if (subjectConfig.Templates.TryGetValue(category, out var template))
            {
                return JsonConvert.SerializeObject(template.Sample, Formatting.Indented);
            }
        }
        return "{}";
    }
}
